ذخیره‌سازی نشست

Hermes Agent از یک پایگاه دادهٔ SQLite (~/.hermes/state.db) برای ماندگارکردن فرادادهٔ نشست، تاریخچهٔ کامل پیام‌ها و پیکربندی مدل در نشست‌های CLI و گیت‌وی استفاده می‌کند. این روش جایگزین رویکرد پیشینِ فایل JSONL جداگانه برای هر نشست شده است.

فایل منبع: hermes_state.py

۱ نمای کلی معماری

~/.hermes/state.db (SQLite, WAL mode)
├── sessions              — فرادادهٔ نشست، شمارش توکن، صورتحساب
├── messages              — تاریخچهٔ کامل پیام‌ها برای هر نشست
├── session_model_usage   — ردیف‌های انتساب مصرف به تفکیک مدل/تسک
├── messages_fts          — جدول مجازی FTS5 (content + tool_name + tool_calls)
├── messages_fts_trigram  — جدول مجازی FTS5 با توکنایزر تری‌گرام (CJK / جستجوی زیررشته)
├── messages_fts_cjk      — جدول مجازی FTS5 با توکنایزر cjk_unicode61
├── state_meta            — جدول فرادادهٔ کلید/مقدار
├── gateway_routing       — فرادادهٔ مسیریابی گیت‌وی
├── compression_locks     — قفل‌گذاری فشرده‌سازی بین‌پروسسی
├── async_delegations     — دفتر ثبت واگذاری‌های ناهمگام
└── schema_version        — جدول تک‌ردیفی برای ردیابی وضعیت مهاجرت

تصمیم‌های کلیدی طراحی:

  • حالت WAL برای خواننده‌های همزمان + یک نویسنده (چندپلتفرمیِ گیت‌وی)
  • جدول مجازی FTS5 برای جستجوی سریع متنی روی همهٔ پیام‌های نشست‌ها
  • تبار نشست از طریق زنجیره‌های parent_session_id (تقسیم‌های ناشی از فشرده‌سازی)
  • برچسب‌گذاری منبع (cli، telegram، discord و...) برای فیلترکردن پلتفرم
  • اجرای دسته‌ای و ترایکتوری‌های RL اینجا ذخیره نمی‌شوند (سیستم‌های جدایی هستند)

۲ طرح‌وارهٔ SQLite

جدول sessions

خلاصه‌شده — فهرست کامل ستون‌های فعلی را در SCHEMA_SQL داخل hermes_state.py ببینید (که ستون‌های فرادادهٔ مسیریابی گیت‌وی مثل session_key، chat_id، chat_type، thread_id، display_name، origin_json، expiry_finalized، فیلدهای فضای کاری cwd / git_branch / git_repo_root، فیلدهای تحویل دست‌به‌دست و شکست فشرده‌سازی، profile_name، rewind_count، archived و pinned را هم شامل می‌شود):

CREATE TABLE IF NOT EXISTS sessions (
    id TEXT PRIMARY KEY,
    source TEXT NOT NULL,
    user_id TEXT,
    model TEXT,
    model_config TEXT,
    system_prompt TEXT,
    parent_session_id TEXT,
    started_at REAL NOT NULL,
    ended_at REAL,
    end_reason TEXT,
    message_count INTEGER DEFAULT 0,
    tool_call_count INTEGER DEFAULT 0,
    input_tokens INTEGER DEFAULT 0,
    output_tokens INTEGER DEFAULT 0,
    cache_read_tokens INTEGER DEFAULT 0,
    cache_write_tokens INTEGER DEFAULT 0,
    reasoning_tokens INTEGER DEFAULT 0,
    billing_provider TEXT,
    billing_base_url TEXT,
    billing_mode TEXT,
    estimated_cost_usd REAL,
    actual_cost_usd REAL,
    cost_status TEXT,
    cost_source TEXT,
    pricing_version TEXT,
    title TEXT,
    api_call_count INTEGER DEFAULT 0,
    -- ... ستون‌های اضافی گیت‌وی/فضای کاری/تحویل دست‌به‌دست/فشرده‌سازی ...
    FOREIGN KEY (parent_session_id) REFERENCES sessions(id)
);

CREATE INDEX IF NOT EXISTS idx_sessions_source ON sessions(source);
CREATE INDEX IF NOT EXISTS idx_sessions_parent ON sessions(parent_session_id);
CREATE INDEX IF NOT EXISTS idx_sessions_started ON sessions(started_at DESC);
CREATE UNIQUE INDEX IF NOT EXISTS idx_sessions_title_unique
    ON sessions(title) WHERE title IS NOT NULL;

جدول messages

خلاصه‌شده — طرح‌وارهٔ کامل شامل effect_disposition، platform_message_id، observed، active، compacted، api_content، display_kind و display_metadata هم هست:

CREATE TABLE IF NOT EXISTS messages (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    session_id TEXT NOT NULL REFERENCES sessions(id),
    role TEXT NOT NULL,
    content TEXT,
    tool_call_id TEXT,
    tool_calls TEXT,
    tool_name TEXT,
    timestamp REAL NOT NULL,
    token_count INTEGER,
    finish_reason TEXT,
    reasoning TEXT,
    reasoning_content TEXT,
    reasoning_details TEXT,
    codex_reasoning_items TEXT,
    codex_message_items TEXT
    -- ... ستون‌های اضافی نمایش/فشرده‌سازی ...
);

CREATE INDEX IF NOT EXISTS idx_messages_session ON messages(session_id, timestamp);
CREATE INDEX IF NOT EXISTS idx_messages_session_id ON messages(session_id, id);

نکته‌ها:

  • tool_calls به‌صورت رشتهٔ JSON ذخیره می‌شود (فهرست سریال‌شدهٔ اشیای فراخوانی ابزار)
  • reasoning_details، codex_reasoning_items و codex_message_items به‌صورت رشتهٔ JSON ذخیره می‌شوند
  • reasoning متن خام استدلال را برای ارائه‌دهنده‌هایی که آن را ارائه می‌کنند ذخیره می‌کند
  • api_content یک سایدکار با وفاداری بایتی است: دقیقاً همان رشتهٔ محتوایی که برای این پیام به API فرستاده شده، وقتی با content فرق دارد (تزریق‌های زودگذر حافظه/پلاگین، بازنویسی‌های پایدار). این فیلد بایت‌های روی سیم را برای بازپخشِ پایدار از نظر کش پرامپت حفظ می‌کند — همان‌طور که فرستاده شده ذخیره می‌شود، به‌جز سروگیت‌های تنها که sqlite3 نمی‌تواند آن‌ها را bind کند و حلقهٔ گفتگو به‌هرحال از هر بار ارسالی پاک‌شان می‌کند. NULL یعنی content عیناً همان‌طور که بود فرستاده شده است.
  • مهرهای زمانی اعداد اعشاری یونیکس‌اپوک هستند (time.time())

جستجوی تمام‌متن FTS5

CREATE VIRTUAL TABLE IF NOT EXISTS messages_fts USING fts5(
    content,
    tool_name,
    tool_calls,
    content='messages',
    content_rowid='id'
);

جدول FTS5 از طریق سه تریگر که روی INSERT، UPDATE و DELETE جدول messages فعال می‌شوند، همگام نگه داشته می‌شود. تریگرهای فعلی روی نشانگرهای fts_rebuild_high_water / fts_rebuild_progress در state_meta گیت شده‌اند (تا بازسازی FTS در پس‌زمینه بتواند بدون اندیس‌گذاری دوباره پیش برود) و هر سه ستون اندیس‌شده را پوشش می‌دهند — SQL دقیق را در SCHEMA_SQL داخل hermes_state.py ببینید.

۳ نسخهٔ طرح‌واره و مهاجرت‌ها

نسخهٔ فعلی طرح‌واره: ۲۳

جدول schema_version یک عدد صحیح واحد ذخیره می‌کند. افزودن ستون‌های ساده به‌شکل اعلانی توسط _reconcile_columns() مدیریت می‌شود (که ستون‌های موجود را با SCHEMA_SQL مقایسه می‌کند و هر کدام را که نبود ADD می‌کند). زنجیرهٔ مبتنی بر نسخه برای مهاجرت‌های داده و تغییرهای اندیس/FTS رزرو شده که به‌شکل اعلانی قابل بیان نیستند:

نسخهتغییر
1طرح‌وارهٔ اولیه (sessions، messages، FTS5)
2افزودن ستون finish_reason به messages
3افزودن ستون title به sessions
4افزودن اندیس یکتا روی title (NULL مجاز است، غیر NULL باید یکتا باشد)
5افزودن ستون‌های صورتحساب: cache_read_tokens، cache_write_tokens، reasoning_tokens، billing_provider، billing_base_url، billing_mode، estimated_cost_usd، actual_cost_usd، cost_status، cost_source، pricing_version
6افزودن ستون‌های استدلال به messages: reasoning، reasoning_details، codex_reasoning_items
7افزودن ستون reasoning_content به messages
8افزودن ستون api_call_count به sessions
9افزودن ستون codex_message_items به messages برای بازپخش شناسه/فاز پیام در Codex Responses
10افزودن جدول مجازی messages_fts_trigram (توکنایزر تری‌گرام برای CJK / جستجوی زیررشته) و پرکردن ردیف‌های موجود
11اندیس‌گذاری مجدد messages_fts و messages_fts_trigram برای پوشش tool_name + tool_calls و تغییر از محتوای خارجی به حالت درون‌خطی؛ حذف تریگرهای قدیمی و پرکردن تک‌تک ردیف‌های پیام
16برچسب‌گذاری ردیف‌های ساب‌ایجنتِ واگذارشده در model_config ($._delegate_from) تا انتخابگرهای نشست پس از یتیم‌شدن بر اثر حذف والد، تمیز بمانند
18تجمیع فرادادهٔ گیت‌وی — پرکردن display_name / origin_json / expiry_finalized از sessions.json
20انتساب مصرف به تفکیک مدل — مقداردهی اولیهٔ ردیف‌های session_model_usage از جمع‌های تجمعی تاریخی هر نشست
22انتساب مصرف در بُعد تسک — بازسازی session_model_usage به‌طوری که ستون task در PRIMARY KEY شرکت کند
23بازطراحی ذخیره‌سازی FTS — جداول FTS با محتوای خارجی که جایگزین نسخه‌های حالت درون‌خطیِ v11 می‌شوند (گذار اختیاری برای پایگاه‌های دادهٔ موجود)

نسخه‌هایی که در بالا فهرست نشده‌اند، افزودن ستونِ اعلانی بوده‌اند که توسط _reconcile_columns() مدیریت می‌شوند (فقط افزایش شمارهٔ نسخه، بدون مهاجرت داده).

افزودن ستون‌های اعلانی از ALTER TABLE ADD COLUMN داخل try/except استفاده می‌کند تا حالت موجودبودنِ ستون مدیریت شود (idempotent). شمارهٔ نسخه بعد از هر بلوک مهاجرت موفق یک واحد بالا می‌رود.

۴ مدیریت تداخل نوشتن

چندین پروسس هرمس (گیت‌وی + نشست‌های CLI + ایجنت‌های ورک‌تری) یک state.db مشترک دارند. کلاس SessionDB تداخل نوشتن را این‌طور مدیریت می‌کند:

  • تایم‌اوت کوتاه SQLite (۱ ثانیه) به‌جای پیش‌فرض ۳۰ ثانیه‌ای
  • تلاش مجدد در سطح برنامه با جیتر تصادفی (۲۰ تا ۱۵۰ میلی‌ثانیه، حداکثر ۱۵ تلاش)
  • تراکنش‌های BEGIN IMMEDIATE برای آشکارسازی تداخل قفل در شروع تراکنش
  • چک‌پوینت‌های دوره‌ای WAL برای هر ۵۰ نوشتار موفق (حالت PASSIVE)

این کار از «اثر کاروان» جلوگیری می‌کند؛ جایی که backoff قطعیِ داخلی SQLite باعث می‌شود همهٔ نویسنده‌های رقیب در همان بازه‌ها تلاش مجدد کنند.

_WRITE_MAX_RETRIES = 15
_WRITE_RETRY_MIN_S = 0.020   # 20ms
_WRITE_RETRY_MAX_S = 0.150   # 150ms
_CHECKPOINT_EVERY_N_WRITES = 50

۶ جستجوی تمام‌متن

متد search_messages() نحو کوئری FTS5 را با پاک‌سازی خودکار ورودی کاربر پشتیبانی می‌کند.

جستجوی پایه

results = db.search_messages("docker deployment")

نحو کوئری FTS5

جستجوی فیلترشده

# مستثناکردن نشست‌های گیت‌وی
results = db.search_messages("bug", exclude_sources=["telegram", "discord"])

قالب نتایج جستجو

هر نتیجه شامل این‌هاست:

  • id، session_id، role، timestamp
  • snippet — قطعهٔ تولیدشده توسط FTS5 با نشانگرهای >>>match<<<
  • context — یک پیام قبل و بعد از مورد منطبق (محتوا به ۲۰۰ کاراکتر بریده می‌شود)
  • source، model، session_started — از نشست والد

متد _sanitize_fts5_query() حالت‌های مرزی را مدیریت می‌کند:

  • گیومه‌های جفت‌نشده و نویسه‌های خاص را حذف می‌کند
  • اصطلاح‌های خط‌تیکه‌دار را داخل گیومه می‌گذارد (chat-send ← "chat-send")
  • عملگرهای بولی سرگردان را حذف می‌کند (hello AND ← hello)

۷ تبار نشست‌ها

نشست‌ها می‌توانند از طریق parent_session_id زنجیره بسازند. این اتفاق وقتی می‌افتد که فشرده‌سازی زمینه، تقسیم نشست را در گیت‌وی فعال کند.

کوئری: یافتن تبار نشست

-- یافتن همهٔ نیاکان یک نشست
WITH RECURSIVE lineage AS (
    SELECT * FROM sessions WHERE id = ?
    UNION ALL
    SELECT s.* FROM sessions s
    JOIN lineage l ON s.id = l.parent_session_id
)
SELECT id, title, started_at, parent_session_id FROM lineage;

-- یافتن همهٔ نسل‌های یک نشست
WITH RECURSIVE descendants AS (
    SELECT * FROM sessions WHERE id = ?
    UNION ALL
    SELECT s.* FROM sessions s
    JOIN descendants d ON s.parent_session_id = d.id
)
SELECT id, title, started_at FROM descendants;

کوئری: نشست‌های اخیر با پیش‌نمایش

SELECT s.*,
    COALESCE(
        (SELECT SUBSTR(m.content, 1, 63)
         FROM messages m
         WHERE m.session_id = s.id AND m.role = 'user' AND m.content IS NOT NULL
         ORDER BY m.timestamp, m.id LIMIT 1),
        ''
    ) AS preview,
    COALESCE(
        (SELECT MAX(m2.timestamp) FROM messages m2 WHERE m2.session_id = s.id),
        s.started_at
    ) AS last_active
FROM sessions s
ORDER BY s.started_at DESC
LIMIT 20;

کوئری: آمار مصرف توکن

-- کل توکن‌ها به تفکیک مدل
SELECT model,
       COUNT(*) as session_count,
       SUM(input_tokens) as total_input,
       SUM(output_tokens) as total_output,
       SUM(estimated_cost_usd) as total_cost
FROM sessions
WHERE model IS NOT NULL
GROUP BY model
ORDER BY total_cost DESC;

-- نشست‌های با بیشترین مصرف توکن
SELECT id, title, model, input_tokens + output_tokens AS total_tokens,
       estimated_cost_usd
FROM sessions
ORDER BY total_tokens DESC
LIMIT 10;

۹ محل پایگاه داده

مسیر پیش‌فرض: ~/.hermes/state.db

این مسیر از hermes_constants.get_hermes_home() به دست می‌آید که به‌طور پیش‌فرض به ~/.hermes/ یا مقدار متغیر محیطی HERMES_HOME حل می‌شود.

فایل پایگاه داده، فایل WAL (state.db-wal) و فایل حافظهٔ مشترک (state.db-shm) همگی در همان پوشه ساخته می‌شوند.
